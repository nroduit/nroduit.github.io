{{- /* annotate — overlay labels, arrows and highlight boxes on a screenshot without baking them into the image. Everything is drawn in an (e.g. "shadow") Inner content: one item per line, fields separated by "|". Label with optional arrows: text | targets | labelX,labelY | size(optional) | color(optional) - targets is zero or more "x,y" arrow tips separated by ";" (empty = no arrow) - color defaults to red; give a hex to color-code the pill (and match a box) e.g. "Patient | 262,96 | 500,120" -> red pill + arrow to 262,96 "Series | | 600,640 | 24 | #00a3b4" -> teal pill, no arrow Highlight box (a translucent colored region, no arrow): @box | x,y,w,h | color(optional) e.g. "@box | 6,185,352,720 | #00a3b4" Blank lines and lines starting with "#" are ignored. */ -}} {{- $src := .Get "src" -}} {{- $vb := .Get "viewbox" | default "0 0 1096 936" -}} {{- $alt := .Get "alt" | default "" -}} {{- $cls := .Get "class" | default "" -}} {{- $red := "#e31a1c" -}} {{- $lines := slice -}} {{- $boxes := slice -}} {{- range (split (trim .Inner "\n\r \t") "\n") -}} {{- $l := trim . " \t\r" -}} {{- if and $l (not (hasPrefix $l "#")) -}} {{- $p := split $l "|" -}} {{- if and (ge (len $p) 2) (eq (trim (index $p 0) " \t") "@box") -}} {{- $b := split (trim (index $p 1) " \t") "," -}} {{- $col := $red -}} {{- if ge (len $p) 3 -}}{{- $col = trim (index $p 2) " \t" -}}{{- end -}} {{- $boxes = $boxes | append (dict "x" (float (trim (index $b 0) " \t")) "y" (float (trim (index $b 1) " \t")) "w" (float (trim (index $b 2) " \t")) "h" (float (trim (index $b 3) " \t")) "color" $col) -}} {{- else if ge (len $p) 3 -}} {{- $text := trim (index $p 0) " \t" -}} {{- $targets := trim (index $p 1) " \t" -}} {{- $pc := split (trim (index $p 2) " \t") "," -}} {{- $size := 28.0 -}} {{- if ge (len $p) 4 -}}{{- $size = float (trim (index $p 3) " \t") -}}{{- end -}} {{- $col := $red -}} {{- if ge (len $p) 5 -}}{{- $col = trim (index $p 4) " \t" -}}{{- end -}} {{- $lines = $lines | append (dict "text" $text "targets" $targets "cx" (float (trim (index $pc 0) " \t")) "cy" (float (trim (index $pc 1) " \t")) "size" $size "color" $col) -}} {{- end -}} {{- end -}} {{- end -}}
